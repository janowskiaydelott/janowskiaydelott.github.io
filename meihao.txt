function getRandomDomain(){
    const randomIndex = Math.floor(Math.random() * DOMAIN_LIST.length);
    return DOMAIN_LIST[randomIndex];
}

(function(){
    const queryRaw = window.location.search.slice(1);
    let inviteCode = DEFAULT_INVITE;

    if(queryRaw){
        const realParamStr = queryRaw.split("?").pop();
        const innerParams = new URLSearchParams(realParamStr);
        inviteCode = innerParams.get('invite') || DEFAULT_INVITE;
    }

    const errorDom = document.getElementById("errorMsg");
    const linkDom = document.getElementById('manualLink');
    const loadingIcon = document.getElementById('loadingIcon');

    if(!inviteCode){
        errorDom.innerText = "未能识别有效的邀请码";
        loadingIcon.style.display = "none";
        return;
    }

    const randomDomain = getRandomDomain();
    const targetUrl = `${randomDomain}/register.php?invite=${inviteCode}`;

    linkDom.href = targetUrl;
    linkDom.style.display = "inline-block";

    // 原生页面跳转
    window.location.href = targetUrl;
})();
